Установка admin пароля OpenSearch: 
export ES_PASS=B.._.7
Запустить OpenSearch: 
docker run -it -p 9200:9200 -p 9600:9600 -e OPENSEARCH_INITIAL_ADMIN_PASSWORD=$ES_PASS -e "discovery.type=single-node"  --name opensearch-node opensearchproject/opensearch:latest
Просмотр nodes: 
curl -X GET "https://localhost:9200/_cat/nodes?v" -ku admin:$ES_PASS
Просмотр plugins: 
curl -X GET "https://localhost:9200/_cat/plugins?v" -ku admin:$ES_PASS
